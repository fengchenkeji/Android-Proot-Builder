# ============================================================================
# Detect target architecture and loader configuration
# ============================================================================
# Extract LOADER_ADDRESS and HAS_LOADER_32BIT from arch.h using the
# preprocessor, mirroring the GNUmakefile approach.
execute_process(
    COMMAND ${CMAKE_C_COMPILER} -E -dM -DNO_LIBC_HEADER
        ${CMAKE_CURRENT_SOURCE_DIR}/arch.h
    OUTPUT_VARIABLE ARCH_DEFINES
    RESULT_VARIABLE ARCH_DEFINES_RESULT
)
if(NOT ARCH_DEFINES_RESULT EQUAL 0)
    message(FATAL_ERROR "Failed to preprocess arch.h")
endif()

# Parse LOADER_ADDRESS
string(REGEX MATCH "#define[ \t]+LOADER_ADDRESS[ \t]+([^ \t\r\n]+)" _ ${ARCH_DEFINES})
set(LOADER_ADDRESS "${CMAKE_MATCH_1}")
if(NOT LOADER_ADDRESS)
    message(FATAL_ERROR "Could not determine LOADER_ADDRESS from arch.h")
endif()
message(STATUS "LOADER_ADDRESS = ${LOADER_ADDRESS}")

# Parse HAS_LOADER_32BIT
string(FIND "${ARCH_DEFINES}" "HAS_LOADER_32BIT" HAS_LOADER_32BIT_POS)
if(HAS_LOADER_32BIT_POS GREATER -1)
    set(HAS_LOADER_32BIT TRUE)
else()
    set(HAS_LOADER_32BIT FALSE)
endif()
message(STATUS "HAS_LOADER_32BIT = ${HAS_LOADER_32BIT}")

# Parse HAS_POKEDATA_WORKAROUND
string(FIND "${ARCH_DEFINES}" "HAS_POKEDATA_WORKAROUND" HAS_POKEDATA_POS)
if(HAS_POKEDATA_POS GREATER -1)
    set(HAS_POKEDATA_WORKAROUND TRUE)
else()
    set(HAS_POKEDATA_WORKAROUND FALSE)
endif()
message(STATUS "HAS_POKEDATA_WORKAROUND = ${HAS_POKEDATA_WORKAROUND}")

# Determine objcopy output target format and binary architecture
# based on the target system processor
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    set(OBJCOPY_OUTPUT_TARGET "elf64-littleaarch64")
    set(OBJCOPY_BINARY_ARCH "aarch64")
    set(LOADER_ARCH_CFLAGS "")
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "arm|armv7|armv8")
    set(OBJCOPY_OUTPUT_TARGET "elf32-littlearm")
    set(OBJCOPY_BINARY_ARCH "arm")
    set(LOADER_ARCH_CFLAGS "")
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64")
    set(OBJCOPY_OUTPUT_TARGET "elf64-x86-64")
    set(OBJCOPY_BINARY_ARCH "i386:x86-64")
    set(LOADER_ARCH_CFLAGS "")
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "i386|i686|x86")
    set(OBJCOPY_OUTPUT_TARGET "elf32-i386")
    set(OBJCOPY_BINARY_ARCH "i386")
    set(LOADER_ARCH_CFLAGS "-mregparm=3")
else()
    message(WARNING "Unknown architecture ${CMAKE_SYSTEM_PROCESSOR}, using auto-detection via objdump")
    set(OBJCOPY_OUTPUT_TARGET "")
    set(OBJCOPY_BINARY_ARCH "")
endif()

# ============================================================================
# Build loader (freestanding, static, no libc)
# ============================================================================
# We build the loader as a separate executable using custom commands because
# it requires -nostdlib -static -ffreestanding which conflicts with the
# shared library build settings.

set(LOADER_SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/loader)
set(LOADER_BUILD_DIR ${CMAKE_CURRENT_BINARY_DIR}/loader)
file(MAKE_DIRECTORY ${LOADER_BUILD_DIR})

# Determine the compiler to use for the loader. For cross-compilation with
# NDK, the same CMAKE_C_COMPILER works. For 32-bit loader on x86_64 host,
# we need -m32; on aarch64 target we need the armv7a cross-compiler.
set(LOADER_COMPILER ${CMAKE_C_COMPILER})
set(LOADER_ASM_COMPILER ${CMAKE_C_COMPILER})

# For the 32-bit loader on aarch64 Android, we need the armv7a compiler
if(HAS_LOADER_32BIT)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
        # Use the armv7a clang from the NDK toolchain
        get_filename_component(NDK_BIN_DIR ${CMAKE_C_COMPILER} DIRECTORY)
        set(LOADER_M32_COMPILER "${NDK_BIN_DIR}/armv7a-linux-androideabi${ANDROID_API_LEVEL}-clang")
        if(NOT EXISTS ${LOADER_M32_COMPILER})
            # Try alternative naming
            set(LOADER_M32_COMPILER "${NDK_BIN_DIR}/arm-linux-androideabi-clang")
        endif()
        set(LOADER_M32_ARCH_CFLAGS "")
        message(STATUS "32-bit loader compiler: ${LOADER_M32_COMPILER}")
    elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64")
        set(LOADER_M32_COMPILER ${CMAKE_C_COMPILER})
        set(LOADER_M32_ARCH_CFLAGS "-m32")
    else()
        message(WARNING "HAS_LOADER_32BIT but no known 32-bit compiler for ${CMAKE_SYSTEM_PROCESSOR}")
        set(HAS_LOADER_32BIT FALSE)
    endif()
endif()

# --- Primary loader ---
set(LOADER_OBJ ${LOADER_BUILD_DIR}/loader.o)
set(LOADER_ASM_OBJ ${LOADER_BUILD_DIR}/assembly.o)
set(LOADER_EXE ${LOADER_BUILD_DIR}/loader.exe)
set(LOADER_WRAPPED_O ${LOADER_BUILD_DIR}/loader-wrapped.o)

# Check if we can auto-detect objcopy format from a compiled object
# For now use the pre-determined values
if(NOT OBJCOPY_OUTPUT_TARGET)
    # Fall back: compile a test object and use objdump
    set(OBJCOPY_OUTPUT_TARGET "binary")
    set(OBJCOPY_BINARY_ARCH "")
endif()

# Build loader objects
add_custom_command(
    OUTPUT ${LOADER_OBJ}
    COMMAND ${LOADER_COMPILER}
        -c ${LOADER_SRC_DIR}/loader.c
        -o ${LOADER_OBJ}
        -fPIC -ffreestanding
        ${LOADER_ARCH_CFLAGS}
        -I${CMAKE_CURRENT_SOURCE_DIR}
        -DNO_LIBC_HEADER
    DEPENDS ${LOADER_SRC_DIR}/loader.c
    COMMENT "Compiling loader.c"
)

add_custom_command(
    OUTPUT ${LOADER_ASM_OBJ}
    COMMAND ${LOADER_ASM_COMPILER}
        -c ${LOADER_SRC_DIR}/assembly.S
        -o ${LOADER_ASM_OBJ}
        -fPIC -ffreestanding
        ${LOADER_ARCH_CFLAGS}
        -I${CMAKE_CURRENT_SOURCE_DIR}
    DEPENDS ${LOADER_SRC_DIR}/assembly.S
    COMMENT "Compiling assembly.S"
)

# Link loader executable
add_custom_command(
    OUTPUT ${LOADER_EXE}
    COMMAND ${LOADER_COMPILER}
        -o ${LOADER_EXE}
        ${LOADER_OBJ} ${LOADER_ASM_OBJ}
        -static -nostdlib
        -Wl,-Ttext=${LOADER_ADDRESS}
        -Wl,-z,noexecstack
    DEPENDS ${LOADER_OBJ} ${LOADER_ASM_OBJ}
    COMMENT "Linking loader executable (Ttext=${LOADER_ADDRESS})"
)

# Strip and objcopy embed. We must run objcopy from the loader build dir
# with a relative filename so the embedded symbols are _binary_loader_exe_*.
add_custom_command(
    OUTPUT ${LOADER_WRAPPED_O}
    COMMAND ${CMAKE_STRIP} loader.exe
    COMMAND ${CMAKE_OBJCOPY}
        --input-target=binary
        --output-target=${OBJCOPY_OUTPUT_TARGET}
        --binary-architecture=${OBJCOPY_BINARY_ARCH}
        loader.exe loader-wrapped.o
    WORKING_DIRECTORY ${LOADER_BUILD_DIR}
    DEPENDS ${LOADER_EXE}
    COMMENT "Embedding loader binary as object"
)

# ============================================================================
# Build 32-bit loader (if needed)
# ============================================================================
set(LOADER_M32_WRAPPED_O "")
if(HAS_LOADER_32BIT)
    set(LOADER_M32_OBJ ${LOADER_BUILD_DIR}/loader-m32.o)
    set(LOADER_M32_ASM_OBJ ${LOADER_BUILD_DIR}/assembly-m32.o)
    set(LOADER_M32_EXE ${LOADER_BUILD_DIR}/loader-m32.exe)
    set(LOADER_M32_WRAPPED_O ${LOADER_BUILD_DIR}/loader-m32-wrapped.o)

    # For the 32-bit loader, we need its own LOADER_ADDRESS
    # Extract -m32 variants from arch.h
    execute_process(
        COMMAND ${LOADER_M32_COMPILER} ${LOADER_M32_ARCH_CFLAGS}
            -E -dM -DNO_LIBC_HEADER
            ${CMAKE_CURRENT_SOURCE_DIR}/arch.h
        OUTPUT_VARIABLE ARCH_M32_DEFINES
        RESULT_VARIABLE ARCH_M32_RESULT
    )
    if(ARCH_M32_RESULT EQUAL 0)
        string(REGEX MATCH "#define[ \t]+LOADER_ADDRESS[ \t]+([^ \t\r\n]+)" _ ${ARCH_M32_DEFINES})
        set(LOADER_M32_ADDRESS "${CMAKE_MATCH_1}")
    else()
        # Fallback: use known 32-bit addresses
        set(LOADER_M32_ADDRESS "0x20000000")
    endif()
    message(STATUS "LOADER_M32_ADDRESS = ${LOADER_M32_ADDRESS}")

    # For the 32-bit loader, the embedded object uses the HOST output format
    # (same as the main library), because the loader binary is just raw data.
    # The --binary-architecture does not affect the output ELF format.
    set(OBJCOPY_M32_OUTPUT_TARGET "${OBJCOPY_OUTPUT_TARGET}")
    set(OBJCOPY_M32_BINARY_ARCH "${OBJCOPY_BINARY_ARCH}")

    add_custom_command(
        OUTPUT ${LOADER_M32_OBJ}
        COMMAND ${LOADER_M32_COMPILER} ${LOADER_M32_ARCH_CFLAGS}
            -c ${LOADER_SRC_DIR}/loader.c
            -o ${LOADER_M32_OBJ}
            -fPIC -ffreestanding
            -I${CMAKE_CURRENT_SOURCE_DIR}
            -DNO_LIBC_HEADER
        DEPENDS ${LOADER_SRC_DIR}/loader.c
        COMMENT "Compiling loader.c (32-bit)"
    )

    add_custom_command(
        OUTPUT ${LOADER_M32_ASM_OBJ}
        COMMAND ${LOADER_M32_COMPILER} ${LOADER_M32_ARCH_CFLAGS}
            -c ${LOADER_SRC_DIR}/assembly.S
            -o ${LOADER_M32_ASM_OBJ}
            -fPIC -ffreestanding
            -I${CMAKE_CURRENT_SOURCE_DIR}
        DEPENDS ${LOADER_SRC_DIR}/assembly.S
        COMMENT "Compiling assembly.S (32-bit)"
    )

    add_custom_command(
        OUTPUT ${LOADER_M32_EXE}
        COMMAND ${LOADER_M32_COMPILER} ${LOADER_M32_ARCH_CFLAGS}
            -o ${LOADER_M32_EXE}
            ${LOADER_M32_OBJ} ${LOADER_M32_ASM_OBJ}
            -static -nostdlib
            -Wl,-Ttext=${LOADER_M32_ADDRESS}
            -Wl,-z,noexecstack
        DEPENDS ${LOADER_M32_OBJ} ${LOADER_M32_ASM_OBJ}
        COMMENT "Linking 32-bit loader executable (Ttext=${LOADER_M32_ADDRESS})"
    )

    add_custom_command(
        OUTPUT ${LOADER_M32_WRAPPED_O}
        COMMAND ${CMAKE_STRIP} loader-m32.exe
        COMMAND ${CMAKE_OBJCOPY}
            --input-target=binary
            --output-target=${OBJCOPY_M32_OUTPUT_TARGET}
            --binary-architecture=${OBJCOPY_M32_BINARY_ARCH}
            loader-m32.exe loader-m32-wrapped.o
        WORKING_DIRECTORY ${LOADER_BUILD_DIR}
        DEPENDS ${LOADER_M32_EXE}
        COMMENT "Embedding 32-bit loader binary as object"
    )
endif()

# ============================================================================
# Build loader-info.c (for HAS_POKEDATA_WORKAROUND targets)
# ============================================================================
set(LOADER_INFO_C "")
if(HAS_POKEDATA_WORKAROUND)
    set(LOADER_INFO_C ${LOADER_BUILD_DIR}/loader-info.c)
    add_custom_command(
        OUTPUT ${LOADER_INFO_C}
        COMMAND readelf -s ${LOADER_EXE} | awk -f ${LOADER_SRC_DIR}/loader-info.awk > ${LOADER_INFO_C}
        DEPENDS ${LOADER_EXE} ${LOADER_SRC_DIR}/loader-info.awk
        COMMENT "Generating loader-info.c"
    )
endif()

# ============================================================================
# Main shared library target: libproot.so
# ============================================================================
set(ALL_PROOT_SOURCES
    ${PROOT_SOURCES}
    ${LOADER_WRAPPED_O}
    ${LOADER_M32_WRAPPED_O}
    ${LOADER_INFO_C}
)

add_library(proot SHARED ${ALL_PROOT_SOURCES})
set_target_properties(proot PROPERTIES
    PREFIX ""
    OUTPUT_NAME "libproot"
    SUFFIX ".so"
    LINK_FLAGS "-Wl,-z,noexecstack -Wl,--export-dynamic"
)
target_include_directories(proot PRIVATE ${TALLOC_DIR})
target_link_libraries(proot PRIVATE
    -Wl,--whole-archive talloc -Wl,--no-whole-archive
)

        -Wl,--whole-archive talloc -Wl,--no-whole-archive
    )
